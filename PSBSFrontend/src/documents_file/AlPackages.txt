CREATE TABLE Packages (
    id INT IDENTITY(1,1) PRIMARY KEY,

    package_name NVARCHAR(100) NOT NULL,
    description NVARCHAR(MAX) NOT NULL,

    coverage_duration_hours INT NOT NULL,
    max_edited_photos INT NOT NULL,

    raw_files_available BIT NOT NULL DEFAULT 0,

    base_price DECIMAL(10,2) NOT NULL,

    created_at DATETIME2 NOT NULL DEFAULT GETDATE(),
    updated_at DATETIME2 NOT NULL DEFAULT GETDATE()
);

CREATE TABLE AddOns (
    id INT IDENTITY(1,1) PRIMARY KEY,

    addon_name NVARCHAR(100) NOT NULL,
    addon_description NVARCHAR(255),
    addon_price DECIMAL(10,2) NOT NULL,

    created_at DATETIME2 NOT NULL DEFAULT GETDATE()
);



CREATE TABLE PackageAddOns (
    id INT IDENTITY(1,1) PRIMARY KEY,

    package_id INT NOT NULL,
    addon_id INT NOT NULL,

    FOREIGN KEY (package_id) REFERENCES Packages(id) ON DELETE CASCADE,
    FOREIGN KEY (addon_id) REFERENCES AddOns(id) ON DELETE CASCADE,

    CONSTRAINT UQ_Package_AddOn UNIQUE (package_id, addon_id)
);





// ERD Diagram code for my project 

// =====================================================
// USERS (Client / Photographer / Admin)
// =====================================================
Table UsersRegistration {
  Id int [pk, increment]
  RegisterAS enum('CLIENT','PHOTOGRAPHER','ADMIN')
  FullName varchar
  Email varchar [unique]
  Phone varchar
  UserName varchar [unique]
  Password varchar
  ConfirmPassword varchar
  Gender varchar
  ProfileImage varchar
  IsAvailable bit              // Photographer availability toggle
  CreatedAT datetime
}

// =====================================================
// PHOTOGRAPHER PROFILE (EXTENSION OF USER)
// =====================================================
Table PhotographerProfiles {
  Id int [pk, increment]
  UserId int [ref: > UsersRegistration.Id]
  AverageRating float
}

// =====================================================
// PHOTOGRAPHER AVAILABILITY (DATE BASED)
// =====================================================
Table PhotographerAvailability {
  Id int [pk, increment]
  PhotographerId int [ref: > UsersRegistration.Id]
  AvailableDate date
  IsAvailable bit
}

// =====================================================
// SERVICE CATEGORIES
// =====================================================
Table ServiceCategories {
  Id int [pk, increment]
  Name varchar                 // Wedding, Birthday, Fashion etc.
  IsActive bit
}

// =====================================================
// PACKAGES / SERVICES
// =====================================================
Table Packages {
  Id int [pk, increment]
  CategoryId int [ref: > ServiceCategories.Id]
  PackageName varchar
  Description text
  CoverageDurationHours int
  MaxEditedPhotos int
  RawFilesAvailable bit
  BasePrice decimal
  CreatedAt datetime
  UpdatedAt datetime
}

// =====================================================
// BOOKINGS
// =====================================================
Table Bookings {
  Id int [pk, increment]
  UserId int                   // Client
  PhotographerId int
  PackageId int
  EventDate date
  EventStartTime time
  EventEndTime time
  EventLocation varchar
  Notes text

  PackageName varchar
  CoverageDurationHours int
  EditedPhotos int
  RawFilesAvailable bit
  Price decimal

  BookingStatus enum('PENDING','CONFIRMED','CANCELLED')
  PaymentStatus enum('PAID','UNPAID')
  PaymentMethod enum('BKASH')

  PhotographerName varchar
  CreatedAt datetime
}

// =====================================================
// PAYMENTS / TRANSACTIONS
// =====================================================
Table Payments {
  Id int [pk, increment]
  BookingId int [ref: > Bookings.Id]
  TransactionId varchar
  Amount decimal
  PaymentMethod enum('BKASH','NAGAD','CARD')
  PaymentStatus enum('PAID','UNPAID')
  CreatedAt datetime
}

// =====================================================
// INVOICES
// =====================================================
Table Invoices {
  Id int [pk, increment]
  BookingId int [ref: > Bookings.Id]
  CustomerContactNumber varchar
  ServicePackageName varchar
  EventDateTime datetime
  TotalAmount decimal
  PaymentStatus varchar
  CreatedAt datetime
}

// =====================================================
// PHOTOGRAPHER PORTFOLIO (ADMIN APPROVAL REQUIRED)
// =====================================================
Table PhotographerPortfolio {
  Id int [pk, increment]
  PhotographerId int [ref: > UsersRegistration.Id]
  Title varchar
  Category varchar
  Description text
  ImageName varchar
  ImageUrl varchar
  IsActive bit
  IsApproved bit              // ❗ Without this = NOT visible to clients
  RejectReason text
  CreatedAt datetime
  UpdatedAt datetime
}

// =====================================================
// REVIEWS & RATINGS
// =====================================================
Table ReviewRatings {
  Id int [pk, increment]
  UserId int [ref: > UsersRegistration.Id]
  PhotographerId int [ref: > UsersRegistration.Id]
  BookingId int [ref: > Bookings.Id]
  Rating int
  ReviewComment text
  IsApproved bit
  IsDeleted bit
  CreatedAt datetime
}

// =====================================================
// RECENT ACTIVITIES (PHOTOGRAPHER DASHBOARD)
// =====================================================
Table RecentActivities {
  Id int [pk, increment]
  PhotographerId int [ref: > UsersRegistration.Id]
  ActivityType varchar
  Message text
  FullName varchar
  CreatedAt datetime
}
